  var connectBtn = $('connectBtn'), connectLbl = $('connectLbl');
  var cv = $('confetti'), ctx = cv.getContext('2d'), particles = [], burstTime = 0;
  function resizeCv(){ cv.width = innerWidth; cv.height = innerHeight; }

  function fireConfetti(ox, oy){
    var cols = ['#8B5CF6', '#14B8A6', '#99F6E4', '#0B3D3B', '#C7B3FF'];
    particles = [];
    for(var i=0; i<40; i++){
      var ang = -Math.PI/2 + (Math.random() - 0.5) * Math.PI * 1.5;
      var spd = 260 + Math.random() * 420;
      particles.push({
        ox: ox, oy: oy, vx: Math.cos(ang)*spd, vy: Math.sin(ang)*spd,
        w: 6 + Math.random()*5, h: 4 + Math.random()*4,
        rot: Math.random()*6.28, spin: (Math.random()-0.5)*12,
        c: cols[i % cols.length]
      });
    }
    burstTime = performance.now();
    (function render(now){
      var t = (now - burstTime) / 1000;
      ctx.clearRect(0, 0, cv.width, cv.height);
      if(t < 2.2){
        particles.forEach(function(p){
          var x = p.ox + p.vx * t * 0.9;
          var y = p.oy + p.vy * t * 0.9 + 0.5 * 850 * t * t;
          var al = Math.max(0, 1 - t / 2.1);
          ctx.save(); ctx.globalAlpha = al; ctx.translate(x, y); ctx.rotate(p.rot + p.spin * t);
          ctx.fillStyle = p.c; ctx.fillRect(-p.w/2, -p.h/2, p.w, p.h);
          ctx.restore();
        });
        requestAnimationFrame(render);
      }
    })(burstTime);
  }
